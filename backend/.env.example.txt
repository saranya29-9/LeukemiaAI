# =========================================================
# MODEL LOADING
# =========================================================

model = None
model_error: Optional[str] = None


def load_model_once():
    global model
    global model_error

    try:
        if not MODEL_PATH.exists():
            model = None
            model_error = f"Model file not found: {MODEL_PATH}"
            print(model_error)
            return

        model = tf.keras.models.load_model(
            str(MODEL_PATH),
            compile=False
        )

        output_shape = model.output_shape

        if isinstance(output_shape, list):
            model = None
            model_error = "Multiple model outputs are not supported."
            return

        if output_shape[-1] != len(CLASS_NAMES):
            model = None
            model_error = (
                f"Model has {output_shape[-1]} outputs. "
                f"Expected {len(CLASS_NAMES)}."
            )
            return

        model_error = None

        print("CNN-LSTM model loaded successfully.")
        print("Model path:", MODEL_PATH)
        print("Model input shape:", model.input_shape)
        print("Model output shape:", model.output_shape)

        model.summary()

    except Exception as exc:
        model = None
        model_error = f"{type(exc).__name__}: {exc}"
        print("Model loading failed:", model_error)


load_model_once()

# =========================================================
# IMAGE PREPROCESSING
# =========================================================

def preprocess_image(image: Image.Image) -> np.ndarray:
    """
    Preprocess an uploaded image for the trained model.

    The model contains a Rescaling(1.0 / 255) layer,
    so this function does not divide pixel values by 255.
    """

    image = image.convert("RGB")

    image = image.resize(
        (IMAGE_SIZE, IMAGE_SIZE)
    )

    image_array = np.asarray(
        image,
        dtype=np.float32
    )

    image_array = np.expand_dims(
        image_array,
        axis=0
    )

    return image_array

@app.post("/predict")
async def predict(
    file: UploadFile = File(...)
):

    # Check content type

    if (
        file.content_type
        and not file.content_type.startswith("image/")
    ):
        raise HTTPException(
            status_code=415,
            detail="Please upload a valid image."
        )

    # Read image with size limit

    image_bytes = await file.read(
        MAX_FILE_BYTES + 1
    )

    if len(image_bytes) > MAX_FILE_BYTES:
        raise HTTPException(
            status_code=413,
            detail="Image size must be 10 MB or less."
        )

    if not image_bytes:
        raise HTTPException(
            status_code=400,
            detail="Uploaded file is empty."
        )

    # Validate image

    try:
        image = Image.open(
            BytesIO(image_bytes)
        )

        image.verify()

        image = Image.open(
            BytesIO(image_bytes)
        ).convert("RGB")

    except (
        UnidentifiedImageError,
        OSError,
        ValueError
    ):
        raise HTTPException(
            status_code=400,
            detail="Uploaded file is not a readable image."
        )

    # Check model availability

    if model is None:
        raise HTTPException(
            status_code=503,
            detail=(
                "Prediction model is unavailable. "
                f"Check /health. {model_error or ''}"
            )
        )

    # Perform prediction

    try:
        processed_image = preprocess_image(image)

        prediction = np.asarray(
            model.predict(
                processed_image,
                verbose=0
            )
        ).reshape(-1)

        # Verify model output

        if prediction.size != len(CLASS_NAMES):
            raise ValueError(
                "Model output does not contain five class scores."
            )

        if not np.all(np.isfinite(prediction)):
            raise ValueError(
                "Model returned invalid numerical values."
            )

        # Interpret output scores.
        #
        # This detects a typical softmax probability distribution.
        # Otherwise, softmax is applied to the output as logits.
        #
        # This is a practical handling method, not a guarantee
        # that model confidence is medically calibrated.

        is_probability_distribution = (
            np.all(prediction >= 0)
            and np.all(prediction <= 1)
            and np.isclose(
                float(np.sum(prediction)),
                1.0,
                atol=0.05
            )
        )

        if is_probability_distribution:

            prediction = prediction / max(
                float(np.sum(prediction)),
                1e-8
            )

        else:

            exp_values = np.exp(
                prediction - np.max(prediction)
            )

            prediction = (
                exp_values / np.sum(exp_values)
            )

        # Find predicted class

        best_index = int(
            np.argmax(prediction)
        )

        confidence = round(
            float(prediction[best_index]) * 100,
            2
        )

        # Prepare class probabilities

        probabilities = {
            CLASS_NAMES[index]: round(
                float(prediction[index]) * 100,
                2
            )
            for index in range(
                len(CLASS_NAMES)
            )
        }

        return {
            "success": True,

            "predicted_class": (
                CLASS_NAMES[best_index]
            ),

            "result": (
                CLASS_NAMES[best_index]
            ),

            "confidence": confidence,

            "class_probabilities": probabilities,

            "disclaimer": (
                "Educational prototype only. "
                "This is not a medical diagnosis. "
                "A qualified hematologist must review "
                "blood-smear findings."
            ),
        }

    except HTTPException:
        raise

    except Exception as exc:
        raise HTTPException(
            status_code=500,
            detail=(
                f"Prediction failed: "
                f"{type(exc).__name__}: {exc}"
            )
        )

@app.post("/chat")
async def chat(
    request: ChatRequest
):

    message = request.message.strip()

    if not message:
        raise HTTPException(
            status_code=400,
            detail="Message cannot be empty."
        )

    reply = get_chatbot_answer(
        message
    )

    return {
        "success": True,
        "mode": "offline",
        "reply": reply,
    }
